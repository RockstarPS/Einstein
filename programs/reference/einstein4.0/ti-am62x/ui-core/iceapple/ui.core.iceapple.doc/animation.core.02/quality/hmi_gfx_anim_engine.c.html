  344:	                  */
  345:	                  fl_anim_def_SP->cb(fl_anim_def_SP->num_frames, fl_anim_def_SP->num_frames);
  346:	               }
  347:	            }
  348:	            else if((((UINT8)p_param_flag) & GFX_ANIM_PAUSE) == GFX_ANIM_PAUSE)
  349:	            {
  350:	               fl_anim_flags_SAP->state = GFX_ANIM_STATE_PAUSED;
  351:	            }
  352:	            else
  353:	            {
  354:	            }
  355:	         }
  356:	         if((((UINT8)p_param_flag) & GFX_ANIM_PLAY) == GFX_ANIM_PLAY)
  357:	         {
  358:	            if(fl_anim_flags_SAP->state == GFX_ANIM_STATE_STOPPED)
  359:	            {
  360:	               if(GFX_ANIM_IS_DIR_FWD(((UINT8)p_param_flag)))
  361:	               {
  362:	                  /*
  363:	                  ** A stopped animation is started from the first
  364:	                  ** frame for forward direction PLAY.
  365:	                  */
  366:	                  fl_anim_flags_SAP->dir    = GFX_ANIM_DIR_FORWARD;
  367:	                  *fl_anim_frame_U8AP       = (UINT8)0;
  368:	                  *fl_anim_telapsed_U32AP   = (UINT32)0;
  369:	
  370:	               }
  371:	               else
  372:	               {
  373:	                  /*
  374:	                  ** A stopped animation is started from the last
  375:	                  ** frame for backward direction PLAY.
  376:	                  ** For backward direction fl_anim_telapsed_U32AP
  377:	                  ** represents the duration pending for the animation
  378:	                  */
  379:	                  fl_anim_flags_SAP->dir    = GFX_ANIM_DIR_BACKWORD;
  380:	                  *fl_anim_frame_U8AP       = (UINT8)(fl_anim_def_SP->num_frames-((UINT32)1));
  381:	                  fl_frames_SP = &fl_anim_def_SP->frames[*fl_anim_frame_U8AP];
  382:	                  *fl_anim_telapsed_U32AP   = fl_frames_SP->duration;
  383:	               }
  384:	               fl_anim_flags_SAP->state = GFX_ANIM_STATE_RUNNING;
  385:	               fl_anim_flags_SAP->start = (UINT8)1;
  386:	               *fl_anim_client_id_U16AP  = (UINT16)p_client_id;
  387:	            #if GFX_ANIM_GOTO_COUNT_ARRAY_SIZE > 0
  388:	               fl_anim_flags_SAP->goto_frame_cnt = (SINT8)(*fl_anim_frame_U8AP);
  389:	            #endif
  390:	            }
  391:	            else if(fl_anim_flags_SAP->state == GFX_ANIM_STATE_PAUSED)
  392:	            {
  393:	               /*
  394:	               ** Accept the run request for a paused animation only if the
  395:	               ** lsh id matches with what is stored and what is given rightnow.
  396:	               ** Also accept change of direction
  397:	               */
  398:	               if(*fl_anim_client_id_U16AP == (UINT16)p_client_id)
  399:	               {
  400:	                  fl_anim_flags_SAP->state = GFX_ANIM_STATE_RUNNING;
  401:	                  if(GFX_ANIM_IS_DIR_FWD(p_param_flag))
  402:	                  {
  403:	                     fl_anim_flags_SAP->dir = GFX_ANIM_DIR_FORWARD;
  404:	                  }
  405:	                  else
  406:	                  {
  407:	                     fl_anim_flags_SAP->dir = GFX_ANIM_DIR_BACKWORD;
  408:	                  }
  409:	               }
  410:	            }
  411:	            else
  412:	            {
  413:	            }
  414:	         }
  415:	         fl_status = fl_anim_flags_SAP->state;
  416:	      }
  417:	   }
  418:	   return(fl_status);
  419:	}

  421:	/*============================================================================
  422:	** Function Name:    hmi_gfx_anim_engine
  423:	** Visibility:       global
  424:	** Description:      Performs execution of active animations
  425:	** Invocation:       By GDG03  
  426:	** Inputs/Outputs:   Derives animation commands and issues to GDG03
  427:	** Critical Section: None.
  428:	** Created:          27-Nov-2012 by EMANOJ1
  429:	** Updated:          27-Nov-2012 by EMANOJ1
  430:	**==========================================================================*/
  431:	void hmi_gfx_anim_engine(void)
  432:	{
  433:	   UINT32             fl_end_frame;
  434:	   UINT32             fl_cur_anim_id;
  435:	   UINT32             fl_anim_timer_cur_U32 = l_anim_timer_cur_U32;
  436:	   UINT32             fl_timer_delta_U32;
  437:	   GFX_ANIM_CONTROL_T fl_anim_ctrl_S;
  438:	   GFX_ANIM_FLAGS_T * fl_flags_SAP;
  439:	   UINT32             fl_retry_count;
  440:	
  441:	   if(l_anim_timer_pre_U32 != fl_anim_timer_cur_U32)
  442:	   {
  443:	      fl_anim_ctrl_S.timer_U32 = (fl_anim_timer_cur_U32-l_anim_timer_pre_U32);
  444:	      l_anim_timer_pre_U32     = fl_anim_timer_cur_U32;
  445:	
  446:	      for(fl_cur_anim_id = (UINT32)0; fl_cur_anim_id < (UINT32)GFX_NUM_ANIMATIONS; fl_cur_anim_id++)
  447:	      {
  448:	         fl_retry_count = (UINT32)2;
  449:	         fl_flags_SAP   = &l_anim_flags_SA[fl_cur_anim_id];
  450:	          
  451:	         while((fl_flags_SAP->state == GFX_ANIM_STATE_RUNNING) && (fl_retry_count > (UINT32)0))
  452:	         {
  453:	            UINT32 fl_frame_U32 = (UINT32)(l_anim_frame_U8A[fl_cur_anim_id]);
  454:	
  455:	            fl_anim_ctrl_S.def_sp     = &lc_animation_table_S[fl_cur_anim_id];
  456:	            fl_anim_ctrl_S.frames_sp  = &fl_anim_ctrl_S.def_sp->frames[fl_frame_U32];
  457:	            fl_anim_ctrl_S.types_sp   = fl_anim_ctrl_S.frames_sp->types;
  458:	            fl_anim_ctrl_S.anim_id    = fl_cur_anim_id;
  459:	            fl_flags_SAP->end_reached = (UINT8)0;
  460:	            l_current_anim_id_U32 = fl_anim_ctrl_S.anim_id;
  461:	
  462:	            if(fl_flags_SAP->start != (UINT32)0)
  463:	            {
  464:	               /*
  465:	               ** If the start flag is set then force time as 0  to make sure
  466:	               ** we execute the first step of the frame.
  467:	               */
  468:	               fl_timer_delta_U32  = (UINT32)0;
  469:	               fl_flags_SAP->start = (UINT8)0;
  470:	               if(fl_flags_SAP->dir == GFX_ANIM_DIR_BACKWORD)
  471:	               {
  472:	                  l_anim_time_elapsed_U32A[fl_cur_anim_id] = fl_anim_ctrl_S.frames_sp->duration;
  473:	                  fl_end_frame = (UINT32)(fl_anim_ctrl_S.def_sp->num_frames-((UINT32)1));
  474:	                  if(l_anim_frame_U8A[fl_cur_anim_id] == fl_end_frame)
  475:	                  {
  476:	                     HMI_GFX_ANIM_SET_CLIENT(l_anim_client_id_U16A[fl_cur_anim_id]);
  477:	                     if(fl_anim_ctrl_S.def_sp->cb != GFX_ANIM02_NULL_PTR)
  478:	                     {
  479:	                        fl_anim_ctrl_S.def_sp->cb(fl_end_frame, fl_anim_ctrl_S.def_sp->num_frames);
  480:	                     }
  481:	                  }
  482:	               }
  483:	               else
  484:	               {
  485:	                  l_anim_time_elapsed_U32A[fl_cur_anim_id] = (UINT32)0;
  486:	                  if(l_anim_frame_U8A[fl_cur_anim_id] == (UINT8)0)
  487:	                  {
  488:	                     HMI_GFX_ANIM_SET_CLIENT(l_anim_client_id_U16A[fl_cur_anim_id]);
  489:	                     if(fl_anim_ctrl_S.def_sp->cb != GFX_ANIM02_NULL_PTR)
  490:	                     {
  491:	                        fl_anim_ctrl_S.def_sp->cb((UINT32)0, fl_anim_ctrl_S.def_sp->num_frames);
  492:	                     }
  493:	                  }
  494:	               }
  495:	            }
  496:	            else
  497:	            {
  498:	               /*
  499:	               ** Animation already started, so pass the actual timer
  500:	               ** values.
  501:	               */
  502:	               fl_timer_delta_U32 = fl_anim_ctrl_S.timer_U32;
  503:	            }
  504:	
  505:	            if(fl_frame_U32 < (UINT32)(fl_anim_ctrl_S.def_sp->num_frames))
  506:	            {
  507:	               fl_anim_ctrl_S.prev_elap_time_U32 = l_anim_time_elapsed_U32A[fl_cur_anim_id];
  508:	               if(fl_flags_SAP->dir == GFX_ANIM_DIR_BACKWORD)
  509:	               {
  510:	                  /*
  511:	                  ** For backward direction the timer is counted downwards.
  512:	                  */
  513:	                  if(l_anim_time_elapsed_U32A[fl_cur_anim_id] > fl_timer_delta_U32)
  514:	                  {
  515:	                     l_anim_time_elapsed_U32A[fl_cur_anim_id] -= fl_timer_delta_U32;
  516:	                  }
  517:	                  else
  518:	                  {
  519:	                     /*
  520:	                     ** Reached the begining (End of backward animation)
  521:	                     ** Clamp the elapsed time and set start bit to 1 so that the
  522:	                     ** next frame starts from 0 time.
  523:	                     */
  524:	                     l_anim_time_elapsed_U32A[fl_cur_anim_id] = (UINT32)0;
  525:	                     fl_flags_SAP->end_reached                = (UINT8)1;
  526:	                  }
  527:	               }
  528:	               else
  529:	               {
  530:	                  /*
  531:	                  ** For backward direction the timer is counted upwards.
  532:	                  */
  533:	                  l_anim_time_elapsed_U32A[fl_cur_anim_id] += fl_timer_delta_U32;
  534:	                  if(l_anim_time_elapsed_U32A[fl_cur_anim_id] > fl_anim_ctrl_S.frames_sp->duration)
  535:	                  {
  536:	                     /*
  537:	                     ** Reached the End (End of forward animation)
  538:	                     ** Clamp the elapsed time and set start bit to 1 so that the
  539:	                     ** next frame starts from 0 time.
  540:	                     */
  541:	                     l_anim_time_elapsed_U32A[fl_cur_anim_id] = fl_anim_ctrl_S.frames_sp->duration;
  542:	                     fl_flags_SAP->end_reached                = (UINT8)1;
  543:	                  }
  544:	               }
  545:	               if(hmi_gfx_anim_engine_run(&fl_anim_ctrl_S) == (UINT32)FALSE)
  546:	               {
  547:	                  /* 
  548:	                  ** animation is not executed due to goto command, so give another try to run the
  549:	                  ** animation (else there will be delay of one frame for the animation to run)
  550:	                  */
  551:	                  fl_retry_count--;
  552:	               }
  553:	               else
  554:	               {
  555:	                  fl_retry_count = (UINT32)0;
  556:	               }
  557:	            }
  558:	            else
  559:	            {
  560:	               fl_retry_count = (UINT32)0;
  561:	            }
  562:	         }
  563:	      }
  564:	      for(fl_cur_anim_id = (UINT32)0; fl_cur_anim_id < (UINT32)GFX_NUM_ANIMATIONS; fl_cur_anim_id++)
  565:	      {
  566:	         fl_flags_SAP = &l_anim_flags_SA[fl_cur_anim_id];
  567:	         if(fl_flags_SAP->state == GFX_ANIM_STATE_RUNNING)
  568:	         {
  569:	            if((((UINT32)l_anim_frame_U8A[fl_cur_anim_id]) >= ((UINT32)lc_animation_table_S[fl_cur_anim_id].num_frames)) ||
  570:	               (l_anim_frame_U8A[fl_cur_anim_id] == (UINT8)GFX_ANIM_MAX_U8))
  571:	            {
  572:	               /*
  573:	               ** Frame number has gone ouside boundary.
  574:	               ** Stop the animation as the configuration is invalid
  575:	               ** or the animation has completed all frames.
  576:	               */
  577:	               hmi_gfx_anim_engine_stop(fl_cur_anim_id);
  578:	            }
  579:	         }
  580:	      }
  581:	   }
  582:	}

  584:	/*============================================================================
  585:	** Function Name:    hmi_gfx_anim_engine_fwd
  586:	** Visibility:       local
  587:	** Description:      Performs execution of animation in forward direction
  588:	** Invocation:       By hmi_gfx_anim_engine  
  589:	** Inputs/Outputs:   Derives animation commands and issues to GDG03
  590:	** Critical Section: None.
  591:	** Created:          27-Nov-2012 by EMANOJ1
  592:	** Updated:          27-Nov-2012 by EMANOJ1
  593:	**==========================================================================*/
  594:	static UINT32 hmi_gfx_anim_engine_run(GFX_ANIM_CONTROL_T * p_anim_ctrl_SP)
  595:	{
  596:	   UINT32 fl_animation_sts = (UINT32)TRUE;
  597:	   UINT32 fl_anim_count;
  598:	   UINT32 fl_stime_U32;
  599:	   UINT32 fl_rate_U32;
  600:	   UINT32 fl_max_step;
  601:	   UINT32 fl_cur_step;
  602:	   UINT32 fl_pre_step;
  603:	   UINT32 fl_cur_anim_id = p_anim_ctrl_SP->anim_id; 
  604:	   #if GFX_ANIM_GOTO_COUNT_ARRAY_SIZE > 0
  605:	   GFX_ANIM_FLAGS_T * fl_anim_flags_SAP = &l_anim_flags_SA[fl_cur_anim_id];
  606:	   #endif
  607:	   UINT32 fl_type;
  608:	#ifdef GFX_DYN_DATA_ANIMATIONS
  609:	   /*
  610:	   ** This flag is added to see if all animations in a frame is completed.
  611:	   ** If all frames are completed and all the animations are dynamic 
  612:	   ** MAL animations then this flag will result in immediately
  613:	   ** stoping the frame.
  614:	   */
  615:	   UINT32 fl_all_anim_done = (UINT32)TRUE;
  616:	#endif
  617:	
  618:	   fl_anim_count = p_anim_ctrl_SP->frames_sp->num_types;
  619:	
  620:	   if(fl_anim_count == (UINT32)0)
  621:	   {
  622:	      /*
  623:	      ** There are no animations in this frame. So move to next frame.
  624:	      */
  625:	      hmi_gfx_anim_engine_next_frame((GFX_ANIM_CONTROL_T const *)p_anim_ctrl_SP);
  626:	   }
  627:	   else
  628:	   {
  629:	      while(fl_anim_count > (UINT32)0)
  630:	      {
  631:	         fl_type = p_anim_ctrl_SP->types_sp->type;
  632:	      #if GFX_ANIM_GOTO_COUNT_ARRAY_SIZE > 0
  633:	         if(fl_type == (UINT32)ANIM_TYPE_GO_TO)
  634:	         {
  635:	            /*
  636:	            ** When a goto element is reached we need to break the loop after
  637:	            ** executing it.
  638:	            */
  639:	            fl_anim_count = (UINT32)0;
  640:	            if(p_anim_ctrl_SP->types_sp->val.go_to == GFX_ANIM02_NULL_PTR)
  641:	            {
  642:	               /*
  643:	               ** If a condition check function is set by user, then the return value of user_goto
  644:	               ** function is set directly as the next frame.\
  645:	               */
  646:	               if(p_anim_ctrl_SP->types_sp->handler.user_goto == GFX_ANIM02_NULL_PTR)
  647:	               {
  648:	                  /*
  649:	                  ** Error condition, handler cannot be NULL.
  650:	                  */
  651:	                  hmi_gfx_anim_engine_stop(fl_cur_anim_id);
  652:	               }
  653:	               else
  654:	               {
  655:	                  hmi_gfx_anim_engine_set_frame(p_anim_ctrl_SP, p_anim_ctrl_SP->types_sp->handler.user_goto(l_anim_frame_U8A[fl_cur_anim_id]));
  656:	               }
  657:	            }
  658:	            else if(p_anim_ctrl_SP->types_sp->handler.goto_count != GFX_ANIM_NULL_HANDLER)
  659:	            {
  660:	               /* 
  661:	               ** We have to reset the goto count variable only for the first run of the
  662:	               ** animation. We do this by using the goto_frame_cnt variable.
  663:	               ** this one increments or decrements only for the first run based
  664:	               ** the direction of animation.
  665:	               */
  666:	               if(fl_anim_flags_SAP->dir == GFX_ANIM_DIR_FORWARD)
  667:	               {
  668:	                  if(fl_anim_flags_SAP->goto_frame_cnt <= l_anim_frame_U8A[fl_cur_anim_id])
  669:	                  {
  670:	                     (*p_anim_ctrl_SP->types_sp->handler.goto_count) = 0;
  671:	                     fl_anim_flags_SAP->goto_frame_cnt = (SINT8)(l_anim_frame_U8A[fl_cur_anim_id]);
  672:	                     fl_anim_flags_SAP->goto_frame_cnt++;
  673:	                  }
  674:	               }
  675:	               else
  676:	               {
  677:	                  if(fl_anim_flags_SAP->goto_frame_cnt >= l_anim_frame_U8A[fl_cur_anim_id])
  678:	                  {
  679:	                     /* 
  680:	                     ** Frame just started so reset the goto counter
  681:	                     */
  682:	                     (*p_anim_ctrl_SP->types_sp->handler.goto_count) = 0;
  683:	                     fl_anim_flags_SAP->goto_frame_cnt = (SINT8)(l_anim_frame_U8A[fl_cur_anim_id]);
  684:	                     fl_anim_flags_SAP->goto_frame_cnt--;
  685:	                  }
  686:	               }
  687:	               /*
  688:	               ** Increment goto_count. As long as the count is within the threshold 
  689:	               ** frame is set to the configured frame.
  690:	               ** Once it reaches set threshold next frame is activated. 
  691:	               */
  692:	               (*p_anim_ctrl_SP->types_sp->handler.goto_count)++;
  693:	               if(*p_anim_ctrl_SP->types_sp->handler.goto_count != 0) 
  694:	               {
  695:	                  if(*p_anim_ctrl_SP->types_sp->handler.goto_count <= p_anim_ctrl_SP->types_sp->val.go_to->repeat_cnt)
  696:	                  {
  697:	                     hmi_gfx_anim_engine_set_frame(p_anim_ctrl_SP, p_anim_ctrl_SP->types_sp->val.go_to->goto_frame);
  698:	                  }
  699:	                  else
  700:	                  {
  701:	                     hmi_gfx_anim_engine_next_frame(p_anim_ctrl_SP);
  702:	                  }
  703:	               }
  704:	               else
  705:	               {
  706:	                  /*
  707:	                  ** Error condition, overflow happened so stop the animation
  708:	                  ** otherwise it may run indefinitely.
  709:	                  */
  710:	                  hmi_gfx_anim_engine_stop(fl_cur_anim_id);
  711:	               }
  712:	            }
  713:	            else
  714:	            {
  715:	               /*
  716:	               ** Error condition, goto count variable pointer cannot be NULL.
  717:	               */
  718:	               hmi_gfx_anim_engine_stop(fl_cur_anim_id);
  719:	            }
  720:	            if(fl_anim_flags_SAP->state != GFX_ANIM_STATE_STOPPED)
  721:	            {
  722:	               if(l_anim_frame_U8A[fl_cur_anim_id] < lc_animation_table_S[fl_cur_anim_id].num_frames)
  723:	               {
  724:	                  /* 
  725:	                  ** goto frame executed and there are still frames to execute 
  726:	                  */
  727:	                  fl_animation_sts = (UINT32)FALSE;
  728:	               #ifdef GFX_DYN_DATA_ANIMATIONS
  729:	                  fl_all_anim_done = (UINT32)FALSE;
  730:	               #endif
  731:	               }
  732:	            }
  733:	         }
  734:	         else
  735:	      #else
  736:	         if(fl_type != (UINT32)ANIM_TYPE_GO_TO)
  737:	      #endif
  738:	         {
  739:	            fl_rate_U32  = p_anim_ctrl_SP->types_sp->rate;
  740:	            fl_stime_U32 = p_anim_ctrl_SP->types_sp->start_time;
  741:	
  742:	            if((l_anim_time_elapsed_U32A[fl_cur_anim_id] >= fl_stime_U32) && (fl_rate_U32 > (UINT32)0))
  743:	            {
  744:	               UINT32 fl_cur_incr_percetage;
  745:	               UINT32 fl_pre_incr_percetage;
  746:	               UINT32 fl_anim_duration = p_anim_ctrl_SP->types_sp->duration;
  747:	            #ifdef GFX_DYN_DATA_ANIMATIONS
  748:	               if(p_anim_ctrl_SP->types_sp->dynamic_data != (UINT8)0)
  749:	               {
  750:	                  /*
  751:	                  ** For dynamic data animations, the duration needs to be dynamically calculated
  752:	                  ** as well.
  753:	                  ** The default configurations tell us duration for a certain range and
  754:	                  ** depending on the current range we have to interpolate to find the 
  755:	                  ** duration
  756:	                  */
  757:	                  fl_anim_duration = hmi_gfx_anim_engine_calculate_dduration(p_anim_ctrl_SP, fl_anim_duration);
  758:	               }
  759:	            #endif   
  760:	               if(fl_anim_duration > (UINT32)0)
  761:	               {
  762:	                  /*
  763:	                  ** Making sure the duration is atleast >= 1tick to have proper
  764:	                  ** start and end for the animation. 
  765:	                  ** Otherwise if the duration is  < rate mSec, the current and
  766:	                  ** previous percentage will evaluate to 100% in the first execution
  767:	                  ** itself and there will not be any animation happening. 
  768:	                  */
  769:	                  fl_rate_U32 = (UINT32)((fl_rate_U32+9u)/10u);
  770:	                  if(fl_anim_duration < fl_rate_U32)
  771:	                  {
  772:	                     fl_anim_duration = fl_rate_U32;
  773:	                  }
  774:	                  /*
  775:	                  ** Find the current incremant percentage
  776:	                  */
  777:	                  fl_stime_U32 = (l_anim_time_elapsed_U32A[fl_cur_anim_id]-fl_stime_U32);
  778:	                  if(fl_stime_U32 > fl_anim_duration)
  779:	                  {
  780:	                     fl_stime_U32 = fl_anim_duration;
  781:	                  }
  782:	
  783:	                  fl_max_step = (UINT32)((fl_anim_duration * 10u)/p_anim_ctrl_SP->types_sp->rate);
  784:	                  fl_cur_step = (UINT32)((fl_stime_U32     * 10u)/p_anim_ctrl_SP->types_sp->rate);
  785:	
  786:	                  if(fl_cur_step == fl_max_step)
  787:	                  {
  788:	                     fl_cur_incr_percetage = 1000u;
  789:	                  }
  790:	                  else
  791:	                  {
  792:	                     fl_cur_incr_percetage = p_anim_ctrl_SP->types_sp->handler.user_ip(fl_cur_step, fl_max_step);
  793:	                  }
  794:	                  /*
  795:	                  ** Find the previous incremant percentage for
  796:	                  ** calculating delta movement incase of XY and WH
  797:	                  */
  798:	                  fl_stime_U32 = p_anim_ctrl_SP->types_sp->start_time;
  799:	                  if(p_anim_ctrl_SP->prev_elap_time_U32 > fl_stime_U32)
  800:	                  {
  801:	                     fl_stime_U32 = (p_anim_ctrl_SP->prev_elap_time_U32-fl_stime_U32);
  802:	                  }
  803:	                  else
  804:	                  {
  805:	                     fl_stime_U32 = (UINT32)0;
