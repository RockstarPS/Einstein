  286:	void msg_SequencerUpdate(void)
  287:	/*============================================================================
  288:	**  Visibility:     Public (.h)
  289:	**----------------------------------------------------------------------------
  290:	**  Description:    Message sequencer update.  Two actions are performed:
  291:	**                  1. Activate a new message when the highest priority message
  292:	**                      changes.
  293:	**                  2. Handle the timeout of the currently selected message.
  294:	**                  3. Based on the options enabled allow low priority message
  295:	**                     to be sent to display.
  296:	**                  4. Special handling when an alert is interrupted by another
  297:	**                     equal priority alert.
  298:	**
  299:	**  Parameters:     None.
  300:	**
  301:	**  Returns:        None.
  302:	**----------------------------------------------------------------------------
  303:	** Created: AFERRIS2 12/Apr/2010
  304:	**==========================================================================*/
  305:	{
  306:	  const MSG_MESSAGE_STRUCT *pMessage;
  307:	  const MSG_MESSAGE_STRUCT *currentMessage = MSG_NULL_PTR;
  308:	  eMessageIdentity MessageID;
  309:	  eMessageIdentity lowCategoryMessageID = MSG_INDEX_UNDEFINED;
  310:	  UINT32 category;
  311:	  UINT16 count;
  312:	  UINT16 currentMessageOptions = 0;
  313:	  UINT8 cycleTimerStatus;
  314:	  UINT8 holdTimerStatus;
  315:	  UINT8 displayTimerStatus;
  316:	  UINT8 dispIndTimerStatus;
  317:	  eMessageIdentity fl_newMessageInCategory; 
  318:	
  319:	
  320:	  #ifdef MSG_CYCLE_OPTION_WITH_TIMEOUT
  321:	    UINT16 displayTimeout;
  322:	    UINT8 msg1secTimerStatus;
  323:	  #endif
  324:	
  325:	  holdTimerStatus = MSG_CHECK_SW_TIMER(seq_HoldTimer);
  326:	  displayTimerStatus = MSG_CHECK_SW_TIMER(seq_DisplayTimer);
  327:	  dispIndTimerStatus = MSG_CHECK_SW_TIMER(seq_DispIndTimer);
  328:	
  329:	
  330:	  /*----------------------------------------------------------------------*/
  331:	  /* Process hold message (if defined)                                    */
  332:	  /*----------------------------------------------------------------------*/
  333:	  if (MSG_INDEX_UNDEFINED != seq_HoldMessageID)
  334:	  {
  335:	    /*------------------------------------------------------------------*/
  336:	    /* Remove message if timer has expired or current message has       */
  337:	    /* changed.                                                         */
  338:	    /*------------------------------------------------------------------*/
  339:	    if (TIMER_RUNNING != holdTimerStatus)
  340:	    {
  341:	      seq_RemoveHoldMessage();
  342:	    }
  343:	  }
  344:	
  345:	  /*----------------------------------------------------------------------*/
  346:	  /* process current message timeouts                                     */
  347:	  /*----------------------------------------------------------------------*/
  348:	  if (MSG_INDEX_UNDEFINED != seq_CurrentMessageID)
  349:	  {
  350:	    pMessage = &messageList[seq_CurrentMessageID];
  351:	
  352:	    /*--------------------------------------------------------------*/
  353:	    /* If the T2 timer is running then the new msg flag is reset    */
  354:	    /* only when the timer expires.                                 */
  355:	    /*--------------------------------------------------------------*/
  356:	    if(TESTBIT(pMessage->Options, MSG_OPTION_DISP_INDICATIONTIME))
  357:	    {
  358:	      if ((TIMER_EXPIRED == dispIndTimerStatus) && (GET_SEMAPHORE(seq_NewMessageFlags, seq_CurrentMessageID)))
  359:	      {
  360:	        #ifdef MSG_OLD_WRNG_REPETITION
  361:	        /*----------------------------------------------------------------------------*/
  362:	        /* If MSG_OPTION_OLD_WRNG_REPEAT is not enabled then clear the newMessageFlag */
  363:	        /*----------------------------------------------------------------------------*/
  364:	        if(!TESTBIT(pMessage->Options, MSG_OPTION_OLD_WRNG_REPEAT))
  365:	        {
  366:	        #endif
  367:	          CLR_SEMAPHORE(seq_NewMessageFlags, seq_CurrentMessageID);
  368:	        #ifdef MSG_OLD_WRNG_REPETITION
  369:	        }
  370:	        /*-----------------------------------------------------------------------*/
  371:	        /* If MSG_OPTION_OLD_WRNG_REPEAT is enabled then new message flag should */
  372:	        /* not be cleared when the value is 0 so invalid array value is checked  */
  373:	        /*-----------------------------------------------------------------------*/
  374:	        else if((TESTBIT(pMessage->Options, MSG_OPTION_OLD_WRNG_REPEAT)) &&
  375:	          (seq_Old_Wrng_RepDuration[seq_CurrentMessageID] == HMI_MSG_OLD_WRNG_REP_INVALID_U16))
  376:	        {
  377:	          /*--------------------------------------------------------------*/
  378:	          /* The warning has now become old so load old warning repeat    */
  379:	          /* duration in the seq_Old_Wrng_RepDuration array               */
  380:	          /*--------------------------------------------------------------*/
  381:	          seq_Old_Wrng_RepDuration[seq_CurrentMessageID] = HMI_MSG_OLD_WRNG_REP_DUR;
  382:	          CLR_SEMAPHORE(seq_NewMessageFlags, seq_CurrentMessageID);
  383:	        }
  384:	        else
  385:	        {
  386:	          /* Do nothing */
  387:	        }
  388:	        #endif
  389:	        seq_Callback(seq_CurrentMessageID, MSG_EVENT_OLD);
  390:	        MSG_CLEAR_SW_TIMER(seq_DispIndTimer);
  391:	      }
  392:	    }
  393:	    /*--------------------------------------------------------------*/
  394:	    /* If the hold timer has expired then message is no longer "NEW"*/
  395:	    /*--------------------------------------------------------------*/
  396:	
  397:	    /* Even for messages for which hold timer is not configured due to timer status check the new message flag gets cleared */
  398:	    /* To avoid this the hold timeout value is also checked before clearing new message flag */
  399:	    else if((TIMER_RUNNING != holdTimerStatus) && ((UINT16)MSG_NO_TIMEOUT != pMessage->HoldTimeout))
  400:	    {
  401:	      CLR_SEMAPHORE(seq_NewMessageFlags, seq_CurrentMessageID);   /* ### QAC WARNING:4130 ### */
  402:	    }
  403:	    else
  404:	    {
  405:	      /*-------------*/
  406:	      /* Do nothing  */
  407:	      /*-------------*/
  408:	    }
  409:	    #ifdef MSG_CYCLE_OPTION_WITH_TIMEOUT
  410:	    /*--------------------------------------------------------------*/
  411:	    /* Read the Seq.current msg options,Current msg may be different*/
  412:	    /* from the message which is in top of Seq Q.                   */
  413:	    /*--------------------------------------------------------------*/
  414:	    pMessage = &messageList[seq_CurrentMessageID];
  415:	    displayTimeout = pMessage->DisplayTimeout;
  416:	    #endif
  417:	
  418:	    /*--------------------------------------------------------------*/
  419:	    /* Message has not changed.  Check for display timeout.         */
  420:	    /*--------------------------------------------------------------*/
  421:	    if((TIMER_EXPIRED == displayTimerStatus)
  422:	      #ifdef MSG_CYCLE_OPTION_WITH_TIMEOUT
  423:	      || ((seq_DsplyTimeList[seq_CurrentMessageID] == 0) && \
  424:	      (0 != displayTimeout) ) /* i.e. NO_TIMEOUT */
  425:	      /*i.e. The 1 sec counter (display time list) is applicable only if message has valid timeout. */
  426:	      #endif
  427:	      )
  428:	
  429:	    {
  430:	      BOOLEAN isDeactivated;
  431:	
  432:	      UINT32 msgIndex = seq_GetViewIndex(seq_CurrentMessageID, MSG_ALL_CATEGORIES);
  433:	      #ifdef DISP_TOUT_NO_TOUT_MSG_BY_PRIORITY
  434:	      /*--------------------------------------------------------------*/
  435:	      /* Since the seq_CurrentMessageID will be removed from the      */
  436:	      /* message queue store the message next to seq_CurrentMessageID */
  437:	      /*--------------------------------------------------------------*/
  438:	      currentMessage = &messageList[seq_CurrentMessageID];
  439:	      category = currentMessage->Category;
  440:	      NextMessageID = seq_GetNextMessage(category, seq_CurrentMessageID);
  441:	      #endif
  442:	      /*----------------------------------------------------------*/
  443:	      /* Remove message if the timer has expired, but don't change*/
  444:	      /* the current message ID.  This will be done below.        */
  445:	      /*----------------------------------------------------------*/
  446:	      isDeactivated = seq_RemoveMessage(seq_CurrentMessageID);
  447:	      seq_Callback(seq_CurrentMessageID, MSG_EVENT_TIMEOUT);
  448:	      if (FALSE != isDeactivated)
  449:	      {
  450:	        seq_UpdateViewer(seq_CurrentMessageID, msgIndex);
  451:	        MSG_SEQUENCER_CLIENT_ADD(seq_CurrentMessageID);
  452:	      }
  453:	    }
  454:	  }
  455:	
  456:	  /*----------------------------------------------------------------------*/
  457:	  /* Check the Cycle Timer status. This might be changed during removal   */
  458:	  /* of HOLD message. Due to this, it moved here.                         */
  459:	  /*----------------------------------------------------------------------*/
  460:	  cycleTimerStatus = MSG_CHECK_SW_TIMER(seq_CycleTimer);
  461:	
  462:	  #ifdef MSG_CYCLE_OPTION_WITH_TIMEOUT
  463:	  /*----------------------------------------------------------------------*/
  464:	  /* Check the onesec message timer status.                               */
  465:	  /*----------------------------------------------------------------------*/
  466:	  msg1secTimerStatus = MSG_CHECK_SW_TIMER(seq_1secTimer);
  467:	  #endif
  468:	
  469:	  /*----------------------------------------------------------------------*/
  470:	  /* Get message to be displayed                                          */
  471:	  /*----------------------------------------------------------------------*/
  472:	  MessageID = seq_MessageList[0];
  473:	
  474:	  if (MSG_INDEX_UNDEFINED != MessageID)
  475:	  {
  476:	    /*------------------------------------------------------------------*/
  477:	    /* Work out which message should be displayed.                      */
  478:	    /*------------------------------------------------------------------*/
  479:	    pMessage = &messageList[MessageID];
  480:	    category = pMessage->Category;
  481:	    /* Coverity fix - Prevention of Invalid seq_CurrentMessageID */
  482:	    if(MSG_INDEX_UNDEFINED != seq_CurrentMessageID)
  483:	    {
  484:	      currentMessage = &messageList[seq_CurrentMessageID];
  485:	      currentMessageOptions = currentMessage->Options;
  486:	    }
  487:	    #ifdef DISP_TOUT_NO_TOUT_MSG_BY_PRIORITY
  488:	    /*------------------------------------------------------------------*/
  489:	    /* When there are no more new messages in the category then check   */
  490:	    /* for cycling option.                                              */
  491:	    /*------------------------------------------------------------------*/
  492:	    if(MSG_INDEX_UNDEFINED == seq_GetNewMessage(category))
  493:	    {
  494:	      #ifdef MSG_ACK_WRNG_REPETITION
  495:	      /* When there are no messages in seq_MessageList and when a msg */
  496:	      /* is added from store queue due to ack warning repetition then */
  497:	      /* seq_CurrentMessageID would be MSG_INDEX_UNDEFINED so consider*/
  498:	      /* the options of seq_MessageList[0]                            */
  499:	      if(MSG_INDEX_UNDEFINED == seq_CurrentMessageID)

 1219:	/*============================================================================
 1220:	**  Visibility:     Public (.h)

 1233:	{
 1234:	  const MSG_MESSAGE_STRUCT *pMessage;

 1236:	  UINT8 msgRemoveStatus;

 1244:	  /*-----------------------------------------------------------------------*/

 1248:	    /*-------------------------------------------------------------------*/

 1250:	    /*-------------------------------------------------------------------*/
 1251:	    if(MessageID == seq_FindActiveMessage(pMessage->Category, MessageID))
 1252:	    {
 1253:	      /*--------------------------------------------------------------*/
 1254:	      /* Message is active. So, send msg remove status as 0           */
 1255:	      /*--------------------------------------------------------------*/
 1256:	    }
 1257:	    else
 1258:	    {

 1299:	      /*--------------------------------------------------------------*/
 1300:	      seq_HoldMessageID = MessageID;
 1301:	    }
 1302:	    else
 1303:	    {
 1304:	      UINT32 msgIndex = seq_GetViewIndex(MessageID, MSG_ALL_CATEGORIES);

 1317:	      }
 1318:	    }
 1319:	
 1320:	    msgRemoveStatus = 1;
 1321:	  }
 1322:	
 1323:	  return(msgRemoveStatus);
 1324:	}

 1477:	**
 1478:	**  Parameters:     Button: Button ID and status
 1479:	**
 1480:	**  Returns:        One of the following:
 1481:	**                  MSG_BTN_NONE        Button event to consumed.
 1482:	**                  MSG_BTN_NOACTION    Button event consumed but no action taken

 1484:	**----------------------------------------------------------------------------
 1485:	** Created: AFERRIS2 14/Apr/2010
 1486:	**==========================================================================*/
 1487:	{
 1488:	  MSG_MESSAGE_STRUCT const * thisMessage;
 1489:	  BOOLEAN isDeactivated;

 1491:	  UINT8 ackAftMinToutTimerStatus;

 1498:	    (MSG_INDEX_UNDEFINED != seq_CurrentMessageID))

 1501:	
 1502:	    /*--------------------------------------------------------------*/

 1504:	    /*--------------------------------------------------------------*/

 1511:	      /*  not expired then button press is ignored.                          */
 1512:	      /*---------------------------------------------------------------------*/
 1513:	      if((!TESTBIT(thisMessage->Options, MSG_OPTION_ACK_AFT_MIN_TOUT)) ||
 1514:	        (TIMER_RUNNING != ackAftMinToutTimerStatus))
 1515:	      {
 1516:	        retValue = MSG_BTN_NOACTION;
 1517:	        /*----------------------------------------------------------*/
 1518:	        /* Remove message but don't change the current message ID.  */
