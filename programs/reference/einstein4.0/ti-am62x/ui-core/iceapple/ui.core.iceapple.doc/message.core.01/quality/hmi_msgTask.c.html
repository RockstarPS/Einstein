   98:	void hm_msg_KSRRobin(void)
   99:	/*============================================================================
  100:	**  Visibility:     Public (hm_msgTask.h)
  101:	**----------------------------------------------------------------------------
  102:	**  Description:    Main module function.  This function is called by the
  103:	**                  scheduler round-robin task list.
  104:	**
  105:	**  Parameters:     None.
  106:	**
  107:	**  Returns:        None.
  108:	**----------------------------------------------------------------------------
  109:	** Created: AFERRIS2 07/Apr/2010
  110:	**==========================================================================*/
  111:	{
  112:	    UINT8 currentIgnitionState;
  113:	    UINT32 mcStatus;
  114:	
  115:	    currentIgnitionState = MSG_GET_IGNITION_STATE();
  116:	
  117:	    msg_ArbitratorUpdate(currentIgnitionState);
  118:	    msg_SequencerUpdate();
  119:	    #ifdef HMI_MSG_ENABLE_NEW_MESSAGE_SEQ_QUEUE_TIMEOUT
  120:	    msg_SequencerNewMsgQTimeoutUpdate();
  121:	    #endif
  122:	    #ifdef MSG_OLD_WRNG_REPETITION
  123:	    msg_SequencerOldWrngRepUpdate();
  124:	    #endif
  125:	    #ifdef MSG_ACK_WRNG_REPETITION
  126:	    msg_StorAckWrngRepUpdate();
  127:	    #endif
  128:	
  129:	    /*--------------------------------------------------------------*/
  130:	    /* Process module status if this is required.                   */
  131:	    /*--------------------------------------------------------------*/
  132:	    if (0UL != msgStatusQualifier)
  133:	    {
  134:	        (void)SIF_EXEC(hm_msg_GetStatus, mcStatus, msgStatusQualifier);
  135:	        if (mcStatus != msgStatus)
  136:	        {
  137:	            /*------------------------------------------------------*/
  138:	            /* Status has changed, so notify user...                */
  139:	            /*------------------------------------------------------*/
  140:	            MSG_STATUS_CHANGE(&mcStatus, msgStatusQualifier);
  141:	        }
  142:	    }
  143:	}

  208:	U32SIF_STANDARD_INTERFACE(hm_msg_GetStatus)
  209:	/*============================================================================
  210:	**  Visibility:     Public (hm_msgTask.h)
  211:	**----------------------------------------------------------------------------
  212:	**  Description:    Get message category status.  This is called by the message
  213:	**                  task whenever the message state is processed, but it can also
  214:	**                  be called by the application when a specific status needs
  215:	**                  to be checked.
  216:	**
  217:	**  Parameters:     *Data:      Returned Status
  218:	**                  Qualifier:  Qualification data (see hmi_msgTask.h)
  219:	**                      B0..B7:     Module selection
  220:	**                      B8..B15     Options
  221:	**                      B16..B31:   Threshold.
  222:	**
  223:	**  Returns:        sif_OK (always)
  224:	**----------------------------------------------------------------------------
  225:	** Created: AFERRIS2 20/Oct/2010
  226:	**==========================================================================*/
  227:	{
  228:	    UINT32 seqStatus = 0;
  229:	    UINT32 storStatus = 0;
  230:	    UINT32 qualifierValue;
  231:	    UINT32 Threshold;
  232:	
  233:	    /*----------------------------------------------------------------------*/
  234:	    /* Get qualifier data (if valid).                                       */
  235:	    /*----------------------------------------------------------------------*/
  236:	    qualifierValue = SIF_GET_QUALIFIER();
  237:	
  238:	     /*-----------------------------------------------------------------*/
  239:	     /* Latch data if requested, so that task can update status without */
  240:	     /* further notification.                                           */
  241:	     /*-----------------------------------------------------------------*/
  242:	    if (TESTBIT(qualifierValue, MSG_CATEGORY_LATCH_QUALIFIER))
  243:	    {
  244:	        CLEARBIT(qualifierValue, MSG_CATEGORY_LATCH_QUALIFIER); /* ### QAC ERROR:3212 ### */
  245:	        msgStatusQualifier = qualifierValue;
  246:	    }
  247:	    /*----------------------------------------------------------------------*/
  248:	    /* Extract threshold from qualifier.                                    */
  249:	    /*----------------------------------------------------------------------*/
  250:	    Threshold = (qualifierValue >> (MSG_THRESHOLD_BITPOS));
  251:	
  252:	    /*----------------------------------------------------------------------*/
  253:	    /* Only check if the caller has provided a valid pointer.               */
  254:	    /*----------------------------------------------------------------------*/
  255:	    if (SIF_ISVALID_DATA())
  256:	    {
  257:	        /*------------------------------------------------------------------*/
  258:	        /* Check SEQ categories...                                          */
  259:	        /*------------------------------------------------------------------*/
  260:	        if (TESTBIT(qualifierValue, MSG_SEQUENCER_CATEGORY))
  261:	        {
  262:	            seqStatus = seq_GetCategories(Threshold);
  263:	        }
  264:	
  265:	        /*------------------------------------------------------------------*/
  266:	        /* Check STOR categories...                                         */
  267:	        /*------------------------------------------------------------------*/
  268:	        if (TESTBIT(qualifierValue, MSG_STORE_CATEGORY))
  269:	        {
  270:	            storStatus = stor_GetCategories(Threshold);
  271:	        }
  272:	
  273:	        /*------------------------------------------------------------------*/
  274:	        /* Combine result and pass to caller.                               */
  275:	        /*------------------------------------------------------------------*/
  276:	        SIF_SET_DATA(seqStatus | storStatus);
  277:	    }
  278:	
  279:	    return sif_OK;
  280:	}
