   77:	void msg_ArbitratorUpdate(UINT8 Ignition)
   78:	/*============================================================================
   79:	**  Visibility:     Public (.h)
   80:	**----------------------------------------------------------------------------
   81:	**  Description:    Check state of all messages and pass on changes to the client.
   82:	**
   83:	**  Parameters:     None.
   84:	**
   85:	**  Returns:        None.
   86:	**----------------------------------------------------------------------------
   87:	** Created: AFERRIS2 12/Apr/2010
   88:	**==========================================================================*/
   89:	{
   90:	  UINT32 i;
   91:	  UINT8  msgStatus;
   92:	  UINT8  msgCurrentState;
   93:	  MSG_MESSAGE_STRUCT const * ptrMessage;
   94:	  SIF_STATUS conditionStatus;
   95:	
   96:	  ptrMessage = &messageList[0];
   97:	
   98:	  for (i=0UL; i < (UINT32)MSG_NUM_MESSAGES; i++)
   99:	  {
  100:	    TEST_INDICATE(i);
  101:	    msgStatus = MSG_STATUS_INACTIVE;
  102:	
  103:	    #ifdef MSG_FREEZE_IGN_MODE
  104:	    /*------------------------------------------------------------------*/
  105:	    /* Check current IGN state to process all the warnings              */
  106:	    /*------------------------------------------------------------------*/
  107:	
  108:	    if (TESTBIT(Ignition, MSG_FREEZE_IGN_MODE) )
  109:	    {
  110:	      /*--------------------------------------------------------------*/
  111:	      /* Check whether the warning is configured in Freeze state      */
  112:	      /*--------------------------------------------------------------*/
  113:	      if(TESTBIT(MSG_FREEZE_IGN_MODE, ptrMessage->ActiveStates) )
  114:	      {
  115:	        /*----------------------------------------------------------*/
  116:	        /* Message is configured in Freeze state, So set warning    */
  117:	        /* check is required as TRUE                                */
  118:	        /*----------------------------------------------------------*/
  119:	        msgCurrentState = TRUE;
  120:	      }
  121:	      else
  122:	      {
  123:	        /*----------------------------------------------------------*/
  124:	        /* Message is NOT configured in Freeze state, So set warning*/
  125:	        /* check is required as FALSE                               */
  126:	        /*----------------------------------------------------------*/
  127:	        msgCurrentState = FALSE;
  128:	      }
  129:	    }
  130:	    else
  131:	    {
  132:	      msgCurrentState = TRUE;
  133:	    }
  134:	
  135:	
  136:	    if(FALSE != msgCurrentState)
  137:	    #endif
  138:	    {
  139:	      /*------------------------------------------------------------------*/
  140:	      /* Check current IGN state is configured for this warning message.  */
  141:	      /*------------------------------------------------------------------*/
  142:	
  143:	      if (TESTBIT(Ignition, ptrMessage->ActiveStates) )
  144:	      {
  145:	        conditionStatus = SIF_EXEC(ptrMessage->CheckActiveCondition,
  146:	          msgCurrentState,
  147:	          ptrMessage->ConditionQualifier);
  148:	        if (sif_OK != conditionStatus)
  149:	        {
  150:	          /*----------------------------------------------------------*/
  151:	          /* If returned status is not "OK" then we can't rely on     */
  152:	          /* the data so assume message request is FALSE              */
  153:	          /*----------------------------------------------------------*/
  154:	          msgCurrentState = MSG_CONDITION_OFF;
  155:	        }
  156:	
  157:	        switch (msgCurrentState)
  158:	        {
  159:	        case MSG_CONDITION_RETRIGGER:
  160:	          /*----------------------------------------------------------*/
  161:	          /* Retrigger the message to restart the display timer to    */
  162:	          /* start the progress bar newly(if the same msg in display) */
  163:	          /*----------------------------------------------------------*/
  164:	          msgStatus = MSG_STATUS_ACTIVE;
  165:	          MSG_ARBITRATOR_RETRIGGER(i);
  166:	          break;
  167:	
  168:	        case MSG_CONDITION_FORCED:
  169:	          /*----------------------------------------------------------*/
  170:	          /* Forcing message to be ACTIVE and behave as though this   */
  171:	          /* is a new message request.                                */
  172:	          /*----------------------------------------------------------*/
  173:	          CLR_SEMAPHORE(arb_MessageStatus, (UINT16)i);    /* ### QAC WARNING:4130 ### */
  174:	          msgStatus = MSG_STATUS_ACTIVE;
  175:	          break;
  176:	
  177:	        case MSG_CONDITION_ACTIVE:
  178:	          /*----------------------------------------------------------*/
  179:	          /* Normal action when a message is ACTIVE.                  */
  180:	          /*----------------------------------------------------------*/
  181:	          msgStatus = MSG_STATUS_ACTIVE;
  182:	          break;
  183:	
  184:	        default:
  185:	          /*----------------------------------------------------------*/
  186:	          /* Nothing to do since the default is to set the            */
  187:	          /* message as INACTIVE                                      */
  188:	          /*----------------------------------------------------------*/
  189:	          break;
  190:	        }
  191:	      }
  192:	
  193:	      /*------------------------------------------------------------------*/
  194:	      /* Notify client if message goes INACTIVE or becomes ACTIVE.        */
  195:	      /*------------------------------------------------------------------*/
  196:	      if (MSG_STATUS_INACTIVE != msgStatus)
  197:	      {
  198:	        if (MSG_STATUS_INACTIVE == GET_SEMAPHORE(arb_MessageStatus, (UINT16)i))
  199:	        {
  200:	          MSG_ARBITRATOR_ADD(i);
  201:	          SET_SEMAPHORE(arb_MessageStatus, (UINT16)i);
  202:	        }
  203:	      }
  204:	      else
  205:	      {
  206:	        if (MSG_STATUS_INACTIVE != GET_SEMAPHORE(arb_MessageStatus, (UINT16)i))
  207:	        {
  208:	          msgStatus = MSG_ARBITRATOR_REMOVE(i, Ignition);
  209:	
  210:	          /*-------------------------------------------------------*/
  211:	          /* Clear the Msg Arbitration status flag only if message */
  212:	          /* is removal is accepted by Sequencer                   */
  213:	          /*-------------------------------------------------------*/
  214:	          if(0 != msgStatus)
  215:	          {
  216:	            CLR_SEMAPHORE(arb_MessageStatus, (UINT16)i);    /* ### QAC WARNING:4130 ### */
  217:	          }
  218:	        }
  219:	      }
  220:	    }
  221:	
  222:	    ptrMessage++;
  223:	  }
  224:	}
