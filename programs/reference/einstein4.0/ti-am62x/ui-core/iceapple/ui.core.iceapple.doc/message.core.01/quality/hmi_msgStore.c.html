  439:	void msg_StoreAdd(eMessageIdentity MessageID)
  440:	/*============================================================================
  441:	**  Visibility:     Public (.h)
  442:	**----------------------------------------------------------------------------
  443:	**  Description:    Adds a new message into the message store.  Action is dependant
  444:	**                  on the message options.  If MSG_OPTION_STORE is set, the
  445:	**                  message is added to the store, otherwise the message is
  446:	**                  transferred directly to the client (if defined).
  447:	**
  448:	**  Parameters:     MessageID:  ID of the message to be added.
  449:	**
  450:	**  Returns:        None.
  451:	**----------------------------------------------------------------------------
  452:	** Created: AFERRIS2 12/Apr/2010
  453:	**==========================================================================*/
  454:	{
  455:	    MSG_MESSAGE_STRUCT const * thisMessage;
  456:	    UINT32 thisIndex;
  457:	    UINT32 endIndex;
  458:	
  459:	    thisMessage = &messageList[MessageID];
  460:	
  461:	    if (TESTBIT(thisMessage->Options, MSG_OPTION_STORE) )
  462:	    {
  463:	        /*------------------------------------------------------------------*/
  464:	        /* Message can be stored.                                           */
  465:	        /*------------------------------------------------------------------*/
  466:	        thisIndex = stor_Categories[thisMessage->Category];
  467:	        endIndex = stor_Categories[thisMessage->Category + 1];
  468:	
  469:	        /*--------------------------------------------------------------*/

  471:	        /*--------------------------------------------------------------*/
  472:	        while (thisIndex < endIndex)
  473:	        {
  474:	            eMessageIdentity aMessage = stor_MessageList[thisIndex];
  475:	            MSG_MESSAGE_STRUCT const *ptrMessage = &messageList[aMessage];
  476:	
  477:	#ifdef STORE_QUEUE_FIFO

  479:	            if (thisMessage->Priority < ptrMessage->Priority)

  484:	            if (thisMessage->Priority <= ptrMessage->Priority)

  560:	            thisIndex++;
  561:	        }
  562:	
  563:	        #ifdef MSG_ACK_WRNG_REPETITION
  564:	        if(TESTBIT(thisMessage->Options, MSG_OPTION_ACK_WRNG_REPEAT))
  565:	        {
  566:	            /*--------------------------------------------------------------*/
  567:	            /* Load invalid value in ack warning repetition duration array  */                                           
  568:	            /*--------------------------------------------------------------*/
  569:	            stor_Ack_Wrng_RepDuration[MessageID] = HMI_MSG_ACK_WRNG_REP_INVALID_U8;
  570:	        }
  571:	        #endif
  572:	    }
  573:	    else
  574:	    {
  575:	        /*------------------------------------------------------------------*/
  576:	        /* Message is not sequenced.  Notify client.                        */
  577:	        /*------------------------------------------------------------------*/
  578:	        MSG_STORE_CLIENT_REMOVE(MessageID);
  579:	    }
  580:	
  581:	    return found;
  582:	}

  615:	static void stor_UpdateViewer(eMessageIdentity MessageID)
  616:	/*============================================================================
  617:	**  Visibility:     Private
  618:	**----------------------------------------------------------------------------
  619:	**  Description:    Notify VIEWER if we need to show a different message.
  620:	**                   (removed message has same or higher priority than the msg
  621:	**                   being viewed). Next message to be viewed can be found using
  622:	**                   the FindNext function.
  623:	**
  624:	**  Parameters:     MessageID:  ID of message being removed.
  625:	**
  626:	**  Returns:        None.
  627:	**----------------------------------------------------------------------------
  628:	** Created: AFERRIS2 27/Apr/2010
  629:	**==========================================================================*/
  630:	{
  631:	#ifdef STORE_QUEUE_VIEW_CURRENT
  632:	
  633:	    /* If the msg queue is empty */
  634:	    if (stor_ViewMessage.Id == MSG_INDEX_UNDEFINED)
  635:	    {
  636:	        /* Get the first message from store list */
  637:	        stor_ViewMessage.Id = (UINT16)stor_MessageList[stor_ViewMessage.Position];
  638:	    }
  639:	    /* If the message currently being pointed is deleted */
  640:	    else if((UINT16)MessageID == stor_ViewMessage.Id)
  641:	    {
  642:	        /* If it is the first message in the queue */
  643:	        if(stor_ViewMessage.Position == 0)
  644:	        {
  645:	            /* Point to the first message from store list */
  646:	            stor_ViewMessage.Id = (UINT16)stor_MessageList[stor_ViewMessage.Position];
  647:	        }
  648:	        /* If the message is in middle or bottom of queue */
  649:	        else
  650:	        {
  651:	            UINT32 index;
  652:	            /* Point to the previous message */
  653:	            index = stor_FindPreviousIndex(stor_ViewMessage.Position, stor_ViewMask);
  654:	            stor_ViewMessage.Id = (UINT16)stor_MessageList[index];
  655:	            stor_ViewMessage.Position = (UINT16)index;
  656:	        }
  657:	    }
  658:	    /* If new msg is added or if old messages are deleted */
  659:	    else
  660:	    {
  661:	        /* Update the position of the currently pointing message */
  662:	        stor_ViewMessage.Position = stor_GetViewIndex((eMessageIdentity)stor_ViewMessage.Id, stor_ViewMask);
  663:	    }
  664:	
  665:	    /* Update the count always */
  666:	    stor_ViewMessage.Count = stor_GetViewCount(stor_ViewMask);
  667:	    MSG_STORE_NOTIFY_VIEWER();
  668:	
  669:	#else
  670:	
  671:	    if ((UINT16)MessageID <= stor_ViewMessage.Id)
  672:	    {
  673:	        UINT32 index;
  674:	
  675:	        /*----------------------------------------------------------*/
  676:	        /* Select next message in list.                             */
  677:	        /*----------------------------------------------------------*/
  678:	        index = stor_FindFirstIndex(stor_ViewMask);
  679:	        stor_ViewMessage.Id = (UINT16)stor_MessageList[index];
  680:	        stor_ViewMessage.Position = stor_GetViewIndex((eMessageIdentity)stor_ViewMessage.Id, stor_ViewMask);
  681:	    }
  682:	
  683:	    stor_ViewMessage.Count = stor_GetViewCount(stor_ViewMask);
  684:	    MSG_STORE_NOTIFY_VIEWER();
  685:	
  686:	#endif
  687:	}

 1054:	void msg_StorAckWrngRepUpdate(void)
 1055:	/*============================================================================
 1056:	**  Visibility:     Public (.h)
 1057:	**----------------------------------------------------------------------------
 1058:	**  Description:    Will check the ack warning repetition duration expiry for
 1059:	**  the messages where MSG_OPTION_ACK_WRNG_REPEAT option is enabled.  If the ack
 1060:	**  warning duration expires then the ack warning is made old by adding the
 1061:	**  message in the sequencer queue.
 1062:	**
 1063:	**  Parameters:     None
 1064:	**
 1065:	**  Returns:        None.
 1066:	**----------------------------------------------------------------------------
 1067:	** Created: adevi  13/Dec/2013
 1068:	**==========================================================================*/
 1069:	{
 1070:	    const MSG_MESSAGE_STRUCT *pMessage;
 1071:	    eMessageIdentity MessageID;
 1072:	    UINT32 MsgTimerStatus_U32;
 1073:	    UINT32 i;
 1074:	
 1075:	    MsgTimerStatus_U32 = MSG_CHECK_SW_TIMER(stor_AckWrng_Rep_Timer);
 1076:	    if ((TIMER_RUNNING != MsgTimerStatus_U32)
 1077:	    #ifdef MSG_ACK_WRNG_REPEAT_TIMER_CONTROL
 1078:	    && (stor_AckWrng_Rep_TimerStatus != ACK_WRNG_REP_TIMER_PAUSE)
 1079:	    #endif
 1080:	       )
 1081:	    {
 1082:	        /*------------------------------------------------------------------*/
 1083:	        /* Restart the timer during the expiry of the scheduled time        */
 1084:	        /*------------------------------------------------------------------*/
 1085:	        MSG_START_SW_TIMER(HMI_MSG_ACK_REP_TIMER_VAL, stor_AckWrng_Rep_Timer);
 1086:	        /*------------------------------------------------------------------*/
 1087:	        /* All the active messages present in stor list which has           */
 1088:	        /* MSG_OPTION_ACK_WRNG_REPEAT option enabled and has valid ack      */
 1089:	        /* warning repetition time is short listed. Ack warning repetition  */
 1090:	        /* duration is decremented every one second.  When the value is 0   */
 1091:	        /* then the message is added to the sequencer queue.                */
 1092:	        /*------------------------------------------------------------------*/
 1093:	        for (i=0UL; i <= (UINT32)MSG_NUM_MESSAGES; i++)
 1094:	        {
 1095:	            MessageID = stor_MessageList[i];
 1096:	            if(MSG_INDEX_UNDEFINED != MessageID)
 1097:	            {
 1098:	                pMessage = &messageList[MessageID];
 1099:	                if(TESTBIT(pMessage->Options, MSG_OPTION_ACK_WRNG_REPEAT))
 1100:	                {
 1101:	                    if(HMI_MSG_ACK_WRNG_REP_INVALID_U8 != stor_Ack_Wrng_RepDuration[MessageID])
 1102:	                    {
 1103:	                        if(stor_Ack_Wrng_RepDuration[MessageID] >= HMI_MSG_VALUE_1)
 1104:	                        {
 1105:	                            stor_Ack_Wrng_RepDuration[MessageID] = (stor_Ack_Wrng_RepDuration[MessageID] - HMI_MSG_VALUE_1);
 1106:	                        }
 1107:	                        else
 1108:	                        {
 1109:	                            stor_Ack_Wrng_RepDuration[MessageID] = 0;
 1110:	                        }
 1111:	                        if(0 == stor_Ack_Wrng_RepDuration[MessageID])
 1112:	                        {
 1113:	                            /* To indicate msg_SequencerAdd that msg is added */
 1114:	                            /* from store queue due to ack warning repetition */
 1115:	                            MsgAddedFromStoreQueue = MSG_ADDED_FROM_STORE_QUEUE;
 1116:	                            /*---------------------------------------------------------------*/
 1117:	                            /* Call sequencer add and add the message to the sequencer queue */
 1118:	                            /*---------------------------------------------------------------*/
 1119:	                            msg_SequencerAdd(MessageID);
 1120:	                            /* Reset the flag once msg_SequencerAdd is executed */
 1121:	                            MsgAddedFromStoreQueue = MSG_NOT_ADDED_FROM_STORE_QUEUE;
 1122:	                            /*------------------------------------------------------------*/
 1123:	                            /* Clear the new message flag since the message should be old */
 1124:	                            /*------------------------------------------------------------*/
 1125:	                            msg_ClearNewMessageFlag(MessageID);
 1126:	                            #ifdef MSG_OLD_WRNG_REPETITION
 1127:	                            /*---------------------------------------------------------*/
 1128:	                            /* Set the old warning repetition duration for the message */
 1129:	                            /*---------------------------------------------------------*/
 1130:	                            msg_SetOldWrngRepDur(MessageID);
 1131:	                            #endif
 1132:	                        }
 1133:	                    }
 1134:	                }
 1135:	            }
 1136:	            else
 1137:	            {
 1138:	                /*----------------------------------------------------------*/
 1139:	                /* Stor list is empty, so terminate the check               */
 1140:	                /*----------------------------------------------------------*/
