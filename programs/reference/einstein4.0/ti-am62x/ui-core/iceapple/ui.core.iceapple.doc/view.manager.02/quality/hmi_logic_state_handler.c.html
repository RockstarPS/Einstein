  152:	        fl_count_U8--;
  153:	        fl_sed_button_status_SP->button_id = LSH_INVALID_BUTTON_ID;

  156:	        fl_sed_button_status_SP->client_id = (UINT8)LSH_INVALID_CLIENT_ID;

  165:	    {
  166:	        #if LSH_NUMBER_OF_LAYERS > 1

  168:	        #else

  175:	            SFhmi_view_manager_02InstanceStruct * fl_dpm_stack_P = fl_dpm_config_SP->vm_ptr;

  184:	                fl_count_U8++;
  185:	            }
  186:	            *fl_dpm_config_SP->hap_ptr = fl_dpm_config_SP->size;
  187:	        }
  188:	    }
  189:	    l_remap_button_id_U16 = LSH_INVALID_BUTTON_ID;
  190:	    l_remap_target_ls_id  = LSH_NUMBER_OF_LOGIC_STATES;
  191:	    l_remap_target_ls_client_id  = (UINT8)LSH_INVALID_CLIENT_ID;
  192:	    UNUSED_VAR(p_init_type_U8);
