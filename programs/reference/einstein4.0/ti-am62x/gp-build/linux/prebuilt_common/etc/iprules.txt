*filter
:INPUT DROP [0:0]
:FORWARD DROP [0:0]
:OUTPUT ACCEPT [0:0]

# Allow all incoming and outgoing traffic on the loopback interface
-A INPUT -i lo -j ACCEPT

# Allow incoming DLT (Port 3490) traffic
-A INPUT -p tcp --dport 3490 -j ACCEPT

# Allow incoming VMF (Port 5432) traffic
-A INPUT -p tcp --dport 5432 -j ACCEPT
COMMIT
